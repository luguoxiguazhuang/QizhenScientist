#!/bin/bash
# M0 continuation orchestrator (takes over from driver.sh after the in-flight
# filter-s42 restart orphaned the driver's wait chain).
#
# Remaining grid (driver phases P5..P10), GPUs 4,5,6,7 only:
#   gen s200 (in flight) + filter s42 (in flight)  -> finish first
#   P5: filter s200 (CPU/judge) || gen s201 (GPUs 4-7)
#   P6: filter s201 (CPU) || student SFT s42 (GPUs 4,5)
#   P7: eval s42 treated (GPUs 4,5) || eval s42 ctrlb (GPUs 6,7)
#   P8: SFT s200 (GPUs 4,5 after filter s200) || SFT s201 (GPUs 6,7 after filter s201)
#   P9: eval s200 (GPUs 4,5) || eval s201 (GPUs 6,7)
#   P10: R018 re-judge stability (CPU) -> R019 verdict (CPU)
# All stage scripts are checkpoint-resumable; this script only sequences them.
set -u
cd /REDACTED/tiaozhanbei/demo/labsafety
PY=/REDACTED/miniconda3/envs/subliminal_mm/bin/python
log() { echo "[$(date -u +%FT%TZ)] $*"; }

GEN200_PID=${1:-}
FILTER42_PID=${2:-}

wait_pid() { # <pid> <desc>
  local pid=$1 desc=$2
  if [ -n "$pid" ] && [ -d "/proc/$pid" ]; then
    log "waiting on $desc (pid $pid)"
    while [ -d "/proc/$pid" ]; do sleep 30; done
    log "$desc (pid $pid) exited"
  else
    log "$desc already exited"
  fi
}

check_pool() { # <seed>
  $PY - "$1" <<'EOF'
import glob, json, sys
seed = sys.argv[1]
for arm in ("treated", "ctrlb"):
    n = 0
    for f in glob.glob(f"runs/gen_s{seed}/{arm}/shard_*.jsonl"):
        with open(f, encoding="utf-8") as fh:
            n += sum(1 for ln in fh if ln.strip())
    assert n >= 11900, f"gen_s{seed} {arm}: only {n} rows"
    print(f"gen_s{seed} {arm}: {n} rows OK", flush=True)
EOF
}

fail() { log "STAGE FAILED: $*"; exit 1; }

# ---------- finish in-flight P4 work ----------
wait_pid "$GEN200_PID" "gen s200"
check_pool 200 || fail "gen s200 pool incomplete"
log "gen s200 pool verified"

wait_pid "$FILTER42_PID" "filter s42"
[ -f runs/filter_s42/filter_stats.json ] || fail "filter s42 produced no filter_stats.json"
[ -f runs/filter_s42/student_data_treated.json ] || fail "filter s42 missing student_data_treated.json"
[ -f runs/filter_s42/student_data_ctrlb.json ] || fail "filter s42 missing student_data_ctrlb.json"
log "filter s42 verified"

# ---------- P5: filter s200 (CPU/judge) || gen s201 (GPUs 4-7) ----------
nohup bash pipeline/run_filter_seed.sh 200 > runs/_continue_filter200.log 2>&1 &
F2=$!
log "P5: filter s200 launched (pid $F2) || gen s201 starting"
bash pipeline/run_gen_seed.sh 201 R016_gen_s201 4,5,6,7 > runs/_continue_gen201.log 2>&1 \
  || fail "gen s201"
check_pool 201 || fail "gen s201 pool incomplete"
log "gen s201 done"

# ---------- P6: filter s201 (CPU) || student SFT s42 (GPUs 4,5) ----------
nohup bash pipeline/run_filter_seed.sh 201 > runs/_continue_filter201.log 2>&1 &
F3=$!
log "P6: filter s201 launched (pid $F3); student SFT s42 starting on GPUs 4,5"
bash pipeline/run_student_sft_seed.sh 42 4,5 > runs/_continue_sft42.log 2>&1 \
  || fail "student SFT s42"
log "student SFT s42 done"
